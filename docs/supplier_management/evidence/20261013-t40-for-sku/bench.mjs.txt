import mysql from "<worktree>/node_modules/mysql2/promise.js";
import { SupplierLookupService } from "<worktree>/server/src/modules/supplier/SupplierLookupService.js";
import { ItemLookupService } from "<worktree>/server/src/modules/item/ItemLookupService.js";
const pool = mysql.createPool({ host: "127.0.0.1", port: 3450, user: process.env.DB_USER, password: process.env.DB_PASSWORD, database: "erp_dev", connectionLimit: 20 });
const database = { query: (s, p) => pool.query(s, p), withTransaction: async (w) => { const c = await pool.getConnection(); try { return await w(c); } finally { c.release(); } } };
const time = { nowMs: () => Date.now() }, logger = { info() {}, warn() {}, error() {} };
const svc = new SupplierLookupService({ database, logger, time, items: new ItemLookupService({ database, logger, time }) });
const [[{ code }]] = await pool.query("SELECT supplier_code AS code FROM suppliers WHERE status='active' AND id % 5 = 0 ORDER BY id DESC LIMIT 1");
const cases = {
  "typical, no q, page 1": [3, {}], "typical, no q, page 50": [3, { page: 50 }], "typical, exact code": [3, { q: code }],
  "typical, name word": [3, { q: "Trading" }], "hot, no q, page 1": [1, {}], "hot, exact code": [1, { q: code }],
  "hot, item code": [1, { q: "SIC-50000" }], "cold, no q": [2, {}]
};
const pct = (a, p) => a[Math.min(a.length - 1, Math.ceil(p / 100 * a.length) - 1)];
for (const [name, [sku, opts]] of Object.entries(cases)) {
  for (let i = 0; i < 3; i++) await svc.listForSku(sku, opts);
  const t = []; let total;
  for (let i = 0; i < 30; i++) { const s = performance.now(); total = (await svc.listForSku(sku, opts)).total; t.push(performance.now() - s); }
  t.sort((a, b) => a - b);
  console.log(name.padEnd(22), "total", String(total).padStart(6), "p50", pct(t, 50).toFixed(0), "p95", pct(t, 95).toFixed(0), "max", t.at(-1).toFixed(0), "ms");
}
// 同 benchmarkSupplierManagement.js 一樣：50 個用戶，每次之後等 0..1000ms；20 秒熱身，量 60 秒。全部都做 for-SKU（最差情況）。
const mix = Object.values(cases); const t = []; let errors = 0;
const start = performance.now(), warm = 20_000, end = warm + 60_000;
await Promise.all(Array.from({ length: 50 }, async (_, u) => { let i = u;
  while (performance.now() - start < end) { const [sku, o] = mix[i++ % mix.length]; const s = performance.now();
    try { await svc.listForSku(sku, o); } catch { errors++; }
    if (s - start >= warm) t.push(performance.now() - s);
    await new Promise((r) => setTimeout(r, Math.floor(Math.random() * 1001))); } }));
t.sort((a, b) => a - b); console.log("50 users think 0-1s   ", "n", t.length, "errors", errors, "p50", pct(t, 50).toFixed(0), "p95", pct(t, 95).toFixed(0), "p99", pct(t, 99).toFixed(0), "max", t.at(-1).toFixed(0), "ms");
await pool.end();
