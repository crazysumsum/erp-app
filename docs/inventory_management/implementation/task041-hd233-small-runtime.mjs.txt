async function treeManifest(directory, exclusions = []) {
  const files = {}, seen = new Set();
  async function visit(name) {
    if (exclusions.some(excluded => name === excluded || name.startsWith(excluded + path.sep))) return;
    const info = await fs.lstat(name);
    if (info.isSymbolicLink()) {
      const realpath = await fs.realpath(name), target = await fs.stat(realpath);
      assert.ok([repo,path.dirname(path.dirname(python)),path.dirname(harness)].some(parent=>realpath===parent||realpath.startsWith(parent+path.sep)),'Unreviewed runtime symlink target');
      files[name] = { symlink: await fs.readlink(name), realpath,
        ...(target.isFile() ? { sha256: sha256(await fs.readFile(realpath)) } : { directory: true }) };
      if (target.isDirectory()) await visit(realpath);
    } else if (info.isDirectory()) {
      const realpath = await fs.realpath(name);
      const children = (await fs.readdir(name)).sort();
      files[name] = { directory: true, children };
      if (seen.has(realpath)) return; seen.add(realpath);
      for (const child of children) await visit(path.join(name, child));
    } else if (info.isFile()) files[name] = { sha256: sha256(await fs.readFile(name)), executable: !!(info.mode & 0o111) };
    else throw new Error('Unsupported runtime input: ' + name);
  }
  await visit(directory); return files;
}
async function resolvedExecutable(name, searchPath = testEnv.PATH) {
  const candidates = name.startsWith('/') ? [name] : searchPath.split(path.delimiter).map(directory => path.join(directory, name));
  for (const candidate of candidates) {
    try {
      await fs.access(candidate, 1); const realpath = await fs.realpath(candidate);
      return { path: candidate, realpath, sha256: sha256(await fs.readFile(realpath)), executable:!!((await fs.stat(realpath)).mode & 0o111),
        symlink: (await fs.lstat(candidate)).isSymbolicLink() ? await fs.readlink(candidate) : null };
    } catch (error) { if (!['ENOENT','EACCES','ENOTDIR'].includes(error.code)) throw error; }
  }
  throw new Error('Executable resolution unknown: '+name);
}

async function bounded(work, ms, label) {
  let timer;
  try { return await Promise.race([work, new Promise((_, reject) => { timer = setTimeout(() => reject(new Error(label)), ms); })]); }
  finally { clearTimeout(timer); }
}
async function launch(executable, args, env, label) {
  const stdoutName = `${label}.stdout.log`, stderrName = `${label}.stderr.log`;
  const stdout = await fs.open(path.join(root, stdoutName), 'wx', 0o600); handles.push(stdout); artifacts.add(stdoutName);
  const stderr = await fs.open(path.join(root, stderrName), 'wx', 0o600); handles.push(stderr); artifacts.add(stderrName);
  report.commands.push({ label, executable, args, cwd: repo, env });
  // The runner may fork a test worker; its fresh process group is owned from spawn.
  const child = spawn(executable, args, { cwd: repo, env, detached: true, stdio: ['ignore', stdout.fd, stderr.fd] });
  const record = { label, child, settled: false, result: null, signals: [], groupTerminal: false }; processes.push(record);
  record.exit = new Promise(resolve => {
    child.once('error', error => { record.spawnError = failure(error); });
    child.once('close', (code, signal) => { record.settled = true; record.result = { code, signal }; resolve(record.result); });
  });
  return record;
}
function groupGone(record) {
  if (!record.child.pid) return record.settled;
  assert.ok(Number.isSafeInteger(record.child.pid) && record.child.pid > 0);
  try { process.kill(-record.child.pid, 0); return false; }
  catch (error) {
    if (error.code === 'ESRCH') return true;
    record.groupError = failure(error); return false;
  }
}
async function groupEnd(record) {
  if (record.groupTerminal) return;
  const deadline = performance.now() + 5000;
  while (!groupGone(record)) {
    assert.ok(performance.now() < deadline, `${record.label} process group disappearance unknown`);
    await delay(25);
  }
  record.groupTerminal = true;
}
async function stop(record) {
  if (record.groupTerminal) return; // A disappeared PGID is never owned again.
  if (record.settled && groupGone(record)) { record.groupTerminal = true; return; }
  cleanupErrors.push(`Normal child completion unproven: ${record.label}`);
  for (const signal of ['SIGTERM', 'SIGKILL']) {
    record.signals.push(signal);
    try {
      if (record.child.pid && !groupGone(record)) process.kill(-record.child.pid, signal);
      await bounded(Promise.all([record.exit, groupEnd(record)]), 5000, `${record.label} ${signal} completion unknown`);
    }
    catch (error) { cleanupErrors.push(error.message); }
    if (record.settled && record.groupTerminal) return;
  }
  record.child.unref();
}
