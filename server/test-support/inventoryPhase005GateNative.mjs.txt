async function bounded(promise, deadline, label, onTimeout = () => {}) {
  const remaining = deadline - performance.now();
  assert.ok(remaining > 0, `${label}: original deadline exhausted`);
  let timer;
  try {
    const value = await Promise.race([promise, new Promise((_, reject) => {
      timer = setTimeout(() => {
        const error = Object.assign(new Error(`${label}: original deadline exhausted`), { code: "HD228_DEADLINE" });
        try { onTimeout(); } catch (cleanupError) { error.cleanupError = cleanupError; }
        reject(error);
      }, remaining);
    })]);
    assert.ok(performance.now() < deadline, `${label}: returned after original deadline`);
    return value;
  } finally { clearTimeout(timer); }
}

async function query(db, label, sql, values = [], deadline = performance.now() + SQL_MS) {
  const started = performance.now(), end = Math.min(deadline, started + SQL_MS);
  const event = { label, startedAtMs: started, deadlineMs: end, status: "IN_PROGRESS" };
  sqlEvents.push(event);
  try {
    const remaining = Math.floor(end - performance.now());
    assert.ok(remaining > 0, `${label}: no SQL dispatch after original deadline`);
    event.dispatchAtMs = performance.now();
    const result = await bounded(db.query({ sql, timeout: remaining }, values), end, label, () => db.destroy());
    event.status = "COMPLETE";
    return result;
  } catch (error) {
    event.status = "FAIL"; event.error = failure(error);
    if (error.code === "PROTOCOL_SEQUENCE_TIMEOUT") db.destroy();
    throw error;
  }
  finally { event.settledAtMs = performance.now(); event.elapsedMs = event.settledAtMs - started; }
}

async function connect(config, schema = config.schema) {
  assert.match(schema, /^erp_inventory_gate_binding_[a-f0-9]{12}$/u);
  const mysql = createRequire(path.join(process.env.HD228_REPO_ROOT, "package.json"))("mysql2");
  const raw = mysql.createConnection({ socketPath: config.socket, user: "root", password: process.env.DB_PASSWORD ?? "",
    database: schema, connectTimeout: SQL_MS, supportBigNumbers: true, bigNumberStrings: true, enableKeepAlive: false });
  // Retain the raw transport before handshake: driver connectTimeout alone ends
  // after first data and does not own an incomplete handshake through cleanup.
  const socket = raw.stream;
  let closed = false;
  const terminal = new Promise(resolve => socket.once("close", () => { closed = true; resolve(); }));
  const connected = new Promise((resolve, reject) => raw.connect(error => error ? reject(error) : resolve()));
  raw.on("error", () => {});
  try {
    await bounded(connected, performance.now() + SQL_MS, "Owned MySQL connect", () => { raw.destroy(); socket.destroy(); });
    return raw.promise();
  } catch (error) {
    raw.destroy(); socket.destroy();
    try { if (!closed) await bounded(terminal, performance.now() + TERMINAL_MS, "Failed handshake socket close", () => socket.destroy()); }
    catch (cleanupError) { error.cleanupError = cleanupError; }
    throw error;
  }
}

async function identity(db) {
  const [[row]] = await query(db, "IDENTITY", "SELECT VERSION() AS version,DATABASE() AS schemaName,@@socket AS socket,@@datadir AS datadir,@@skip_networking AS isolated,@@performance_schema AS instrumented,CONNECTION_ID() AS id,CURRENT_USER() AS account");
  return { mysql: { version: row.version, socket: row.socket, datadir: row.datadir.replace(/\/$/u, ""), isolated: Number(row.isolated) },
    schema: row.schemaName, threadId: Number(row.id), account: row.account, performanceSchema: Number(row.instrumented) };
}

async function closeDatabase(db, deadline) {
  const socket = db.connection.stream, observed = { endEvent: false, closeEvent: false, hadError: null };
  socket.once("end", () => { observed.endEvent = true; });
  const closed = new Promise(resolve => {
    socket.once("close", hadError => { observed.closeEvent = true; observed.hadError = hadError; resolve(); });
  });
  await bounded(db.end(), deadline, "Owned MySQL end", () => db.destroy());
  await bounded(closed, deadline, "Owned MySQL socket close event", () => socket.destroy());
  assert.equal(observed.closeEvent, true); assert.equal(observed.hadError, false);
  return observed;
}

async function childMain() {
  const config = configuration(), nonce = process.argv[3], incarnation = process.argv[4];
  assert.match(nonce ?? "", /^[a-f0-9]{32}$/u); assert.match(incarnation ?? "", /^[a-f0-9-]{36}$/u);
  const connections = new Map();
  let queue = Promise.resolve(), expired = false;
  const send = message => new Promise((resolve, reject) => {
    process.send({ nonce, incarnation, ...message }, error => { if (error) reject(error); else resolve(); });
  });
  async function closeOwned(label, deadline) {
    const owned = connections.get(label); assert.ok(owned, "Only a registered owned connection may close");
    let originalError, socket;
    try {
      if (owned.transaction) await query(owned.db, "ROLLBACK", "ROLLBACK", [], Math.min(deadline, owned.transactionDeadline));
    } catch (error) { originalError = error; }
    finally { clearTimeout(owned.watchdog); }
    try { socket = await closeDatabase(owned.db, deadline); connections.delete(label); }
    catch (error) { if (originalError) originalError.cleanupError = error; else originalError = error; }
    if (originalError) throw originalError;
    return { threadId: owned.identity.threadId, socket };
  }
  process.on("message", message => {
    queue = queue.then(async () => {
      const offset = sqlEvents.length;
      try {
        assert.equal(message.nonce, nonce); assert.equal(message.incarnation, incarnation);
        assert.match(message.requestId ?? "", /^[a-f0-9-]{36}$/u);
        if (!["cleanup", "shutdown"].includes(message.action)) assert.equal(expired, false, "Original owned transaction timeout remains a failure");
        let value;
        if (message.action === "open") {
          assert.ok(!connections.has(message.label));
          const db = await connect(config, message.schema), owned = { db, transaction: false };
          connections.set(message.label, owned);
          owned.identity = await identity(db); checkIdentity(owned.identity, config, message.schema);
          value = owned.identity;
        } else if (message.action === "begin") {
          assert.match(message.table, /^gate_probe_[a-f0-9]{12}$/u);
          const owned = connections.get(message.label); assert.ok(owned && !owned.transaction);
          const deadline = performance.now() + TRANSACTION_MS;
          owned.transaction = true; owned.transactionDeadline = deadline;
          owned.watchdog = setTimeout(() => {
            expired = true;
            queue = queue.then(async () => {
              const timeout = Object.assign(new Error("Original transaction20s exhausted"), { code: "HD228_TRANSACTION_TIMEOUT" });
              try { await closeOwned(message.label, performance.now() + TERMINAL_MS); }
              catch (cleanupError) { timeout.cleanupError = cleanupError; }
              await send({ event: "transactionExpired", error: failure(timeout) });
            });
          }, TRANSACTION_MS);
          await query(owned.db, "BEGIN", "START TRANSACTION", [], deadline);
          await query(owned.db, "OWNED_INSERT", `INSERT INTO \`${message.table}\` (id,owner_nonce) VALUES (?,?)`, [message.id, nonce], deadline);
          const [[row]] = await query(owned.db, "OWNER_READ", `SELECT COUNT(*) AS n FROM \`${message.table}\` WHERE id=? AND owner_nonce=?`, [message.id, nonce], deadline);
          assert.equal(Number(row.n), 1);
          value = { ownerRows: Number(row.n), transactionStartedAtMs: deadline - TRANSACTION_MS };
        } else if (message.action === "close") {
          value = await closeOwned(message.label, performance.now() + Math.min(TERMINAL_MS, message.remainingMs));
        } else if (message.action === "loopback") {
          assert.ok(Number.isInteger(message.port) && message.port > 0 && message.port <= 65535);
          const deadline = performance.now() + TERMINAL_MS, socket = net.createConnection({ host: "127.0.0.1", port: message.port });
          const observed = { connected: false, endEvent: false, closeEvent: false, hadError: null, response: "" };
          socket.setEncoding("utf8");
          socket.on("data", chunk => { observed.response += chunk; assert.ok(observed.response.length <= 128); });
          socket.once("end", () => { observed.endEvent = true; });
          const closed = new Promise((resolve, reject) => {
            socket.once("error", reject);
            socket.once("close", hadError => { observed.closeEvent = true; observed.hadError = hadError; resolve(); });
          });
          await bounded(once(socket, "connect"), deadline, "Owned loopback connect", () => socket.destroy());
          observed.connected = true; socket.end(nonce);
          await bounded(closed, deadline, "Owned loopback close event", () => socket.destroy());
          assert.deepEqual([observed.endEvent, observed.closeEvent, observed.hadError, observed.response], [true, true, false, `ACK:${nonce}`]);
          value = observed;
        } else if (message.action === "shutdown") {
          assert.equal(connections.size, 0, "Normal worker exit requires every owned DB socket closed");
          await send({ requestId: message.requestId, value: { ownConnections: 0 }, sqlEvents: sqlEvents.slice(offset) });
          process.disconnect(); return;
        } else if (message.action === "cleanup") {
          const deadline = performance.now() + TERMINAL_MS, closed = [];
          for (const label of connections.keys()) closed.push(await closeOwned(label, deadline));
          value = closed;
        } else assert.fail("Unrecognized resource-probe command");
        await send({ requestId: message.requestId, value, sqlEvents: sqlEvents.slice(offset) });
      } catch (error) {
        process.exitCode = 1;
        await send({ requestId: message.requestId, error: failure(error), sqlEvents: sqlEvents.slice(offset) });
      }
    }).catch(() => { process.exitCode = 1; process.disconnect(); });
  });
  await send({ event: "ready", pid: process.pid });
}

      assert.throws(() => buildMergeRecord({ before, after: changed, outcome: { exitCode: 0, signal: null }, lcovSha256: changedHash, parts }), /inputs changed/u);
      assert.equal(sha(await fs.readFile(recorderPath)), recorderHash);
      return { claims: "SYNTHETIC_VERIFIER_ONLY_NO_FULL_NATIVE_OR_GLOBAL_CI_PASS", recorderHash, lcovSha256,
        sourceFileSha256: before.inputs.sourceFile, partHashes: parts.map(part => part.sha256), changedCopyRejected: true };
    } finally {
      await fs.rm(directory, { recursive: true });
      await assert.rejects(fs.stat(directory), { code: "ENOENT" });
    }
  });

  if (enabled) after(async () => {
    async function cleanup(label, body) {
      try { await body(); } catch (error) { report.workerCleanupErrors.push({ label, error: failure(error) }); }
    }
    if (activeBody) await cleanup("CANCELLED_BODY_SETTLEMENT", () => bounded(activeBody,
      performance.now() + TERMINAL_MS, "Cancelled native body settlement"));
    if (worker && !exitValue) {
      await cleanup("OWNED_WORKER_CLOSE", async () => {
        // Cleanup cannot convert the original failing test into a passing case.
        await request("cleanup");
        const start = performance.now(), deadline = start + TERMINAL_MS;
        for (const entry of owned.values()) await terminal(entry, deadline, start);
        await request("shutdown", {}, deadline); await bounded(workerExit, deadline, "Cleanup child exit");
      });
      if (!exitValue) {
        worker.kill("SIGTERM");
        await cleanup("OWNED_WORKER_FALLBACK_EXIT", () => bounded(workerExit, performance.now() + 5000, "Owned child SIGTERM exit"));
      }
    }
    if (serverSocket && !serverSocket.destroyed) serverSocket.destroy();
    if (server?.listening) await cleanup("OWNED_LISTENER_CLOSE", async () => {
      const closed = once(server, "close"); server.close(); await bounded(closed, performance.now() + 5000, "Failure cleanup listener close");
    });
    if (monitor) {
      await cleanup("OWNED_SQL_BEFORE_DDL", async () => {
        const start = performance.now(), deadline = start + TERMINAL_MS;
        for (const entry of owned.values()) await terminal(entry, deadline, start);
        if (mismatchOwned) { await query(monitor, "CLEANUP_OWN_MISMATCH_SCHEMA", `DROP DATABASE \`${mismatchSchema}\``); mismatchOwned = false; }
        if (tableOwned) { await query(monitor, "DROP_OWN_TABLE", `DROP TABLE \`${table}\``); tableOwned = false; }
      });
      await cleanup("OWNED_MONITOR_CLOSE", async () => { report.monitorSocket = await closeDatabase(monitor, performance.now() + TERMINAL_MS); });
    }
    await cleanup("SOURCE_AFTER", async () => { report.sourceAfter = await sourceHashes(); assert.deepEqual(report.sourceAfter, report.sourceBefore); });
    report.status = report.cases.length === 8 && report.cases.every(row => row.outcome === "PASS") && !report.workerCleanupErrors.length && report.cleanup === "CHILD_EXIT_AND_SQL_TERMINAL_CONFIRMED" ? "PASS" : "FAIL";
    report.outerCleanup = "NOT_CLAIMED_REQUIRES_COORDINATOR_RECEIPT";
    await fs.writeFile(reportPath, JSON.stringify(report, null, 2) + "\n", { flag: "wx", mode: 0o600 });
    assert.equal(report.status, "PASS", "Original case failures and bounded cleanup errors remain failures");
  }, { timeout: 120000 });
}
